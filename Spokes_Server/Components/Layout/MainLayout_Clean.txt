    private bool _drawerOpen = true;
    private bool _isDarkMode = true;
    private MudTheme? _theme = null;
    private Employee? _currentUser;
    private CompanyProfile? Profile;
    protected override async Task OnInitializedAsync()
    {
        Profile = Db.CompanyProfile.Get();
        // 1. Setup Theme
        _theme = new()
            {
                PaletteLight = _lightPalette,
                PaletteDark = _darkPalette,
                LayoutProperties = new LayoutProperties()
            };
        // 2. Load User Preference
        var authState = await AuthStateProvider.GetAuthenticationStateAsync();
        var user = authState.User;
        var sub = user.FindFirst("sub")?.Value
        ?? user.FindFirst(System.Security.Claims.ClaimTypes.NameIdentifier)?.Value;
        if (sub != null)
        {
            // Look up the user in our DB
            // Look up the current user via OpenIdAccount (fresh from database)
            // This ensures we always have the latest link status, not stale claims
            var openIdAccount = Db.OpenIdAccounts.GetBySub(sub);
            if (openIdAccount != null && !string.IsNullOrEmpty(openIdAccount.LinkedEmployeeId))
            {
                _currentUser = Db.Employees.GetById(openIdAccount.LinkedEmployeeId);
            }
            else
            {
                // Fallback to old OidcSub lookup for backwards compatibility
                _currentUser = Db.Employees.GetByOidcSub(sub);
            }
            // Redirect to access-pending if authenticated but no employee linked
            if (user.Identity?.IsAuthenticated == true && _currentUser == null)
            {
                var currentUri = Nav.Uri;
                if (!currentUri.Contains("/access-pending"))
                {
                    Nav.NavigateTo("/access-pending", replace: true);
                    return;
                }
            }
            if (_currentUser != null)
            {
                _isDarkMode = _currentUser.IsDarkMode;
            }
        }
    }
    private void DrawerToggle()
    {
        _drawerOpen = !_drawerOpen;
    }
    private void DarkModeToggle()
    {
        _isDarkMode = !_isDarkMode;
        // 3. Save Preference to Disk
        if (_currentUser != null)
        {
            // Re-fetch the latest version of the user to ensure we don't overwrite other changes
            var freshUser = Db.Employees.GetById(_currentUser.Id);
            if (freshUser != null)
            {
                freshUser.IsDarkMode = _isDarkMode;
                Db.Employees.Save(freshUser);
            }
        }
    }
    private readonly PaletteLight _lightPalette = new()
